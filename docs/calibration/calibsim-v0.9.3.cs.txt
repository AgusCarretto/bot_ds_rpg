int N = args.Length > 0 ? int.Parse(args[0]) : 600;
string[] Classes = ["Guerrero", "Hechicero", "Ninja", "Arquero"];
string Family(string cls) => cls switch { "Guerrero" => "Espadas", "Ninja" => "Dagas", "Arquero" => "Arcos", _ => "Grimorios" };

string conn = File.ReadAllLines(@"C:\Users\Soporte_IT\Desktop\Proyectos AC\bot_ds_rpg\.env")
    .Where(l => l.StartsWith("Postgres__ConnectionString=")).Select(l => l["Postgres__ConnectionString=".Length..].Trim().Trim('"')).First();
await using var db = new NpgsqlConnection(conn);
var rows = (await db.QueryAsync<(int rank, string name, string kind, int hpMin, int hpMax, int dMin, int dMax, int gold, int xp)>(
    @"SELECT (DENSE_RANK() OVER (ORDER BY z.min_level))::int, m.name, CASE WHEN m.is_boss THEN 'boss' WHEN m.is_travel THEN 'travel' ELSE 'hunt' END,
             m.min_hp, m.max_hp, m.min_damage, m.max_damage, m.gold_reward, m.xp_reward
      FROM monsters m JOIN zones z USING (zone_id) ORDER BY z.min_level, m.name")).ToList();
Mon[] Of(int rank, string kind) => rows.Where(r => r.rank == rank && r.kind == kind).Select(r => new Mon(r.name, r.hpMin, r.hpMax, r.dMin, r.dMax, r.gold, r.xp)).ToArray();

// Equipo base por zona (rework_drops_and_recipes.sql): arma ATQ y amuleto DEF; niveles (entrada, medio, del jefe) de cada zona (los de la escalera).
int[] W = [9, 20, 32, 50, 80], A = [10, 18, 30, 46, 75];
(int In, int Mid, int Boss)[] L = [(1, 3, 5), (5, 7, 10), (10, 12, 15), (15, 17, 20), (20, 22, 25)];

Console.WriteLine($"=== 1) EQUIPO PROPIO de la zona, tier en arma y amuleto (n = {N} por clase y monstruo) ===");
Console.WriteLine("Objetivo calibrado en tier 0: comunes cuestan ~13-18% de la vida; jefe ~16% de derrota con el equipo propio.\n");
Console.WriteLine("Zona | tier | comunes: %vida perdida / %derrota | viaje: %vida / %derrota | jefe: %vida / %derrota");
for (int z = 1; z <= 5; z++)

    {

        Console.WriteLine($"  Z{z} |  {tier}   | {c.Lost,5:F1}% / {c.Lose,5:F1}%       | {t.Lost,5:F1}% / {t.Lose,5:F1}%    | {b.Lost,5:F1}% / {b.Lose,5:F1}%");

}

Console.WriteLine("\n=== 2) ENTRAR a la zona Z con el equipo de la Z-1 (tier en ambos) ===");

Console.WriteLine("Zona | tier | comunes: %vida / %derrota | jefe: %vida / %derrota");

Console.WriteLine("\n=== 3) ¿Con cuánta vida es seguro pelear? % de DERROTA según la vida con la que empezás (comunes de la zona, equipo propio sin encantar) ===");

double[] fracs = [1.0, 0.75, 0.5, 0.35, 0.25, 0.15];

for (int z = 1; z <= 5; z++)

    var boss = Of(z, "boss")[0]; int lvl = L[z - 1].Boss; int req = LevelingCalculator.RequiredXpForLevel(lvl);
    double winXp = Enumerable.Range(0, 4000).Average(_ => CombatRewardCalculator.RollBossReward(lvl, boss.GoldBonus, boss.XpBonus, true).Xp);
    double xpHunt = Of(z, "hunt").Average(m => Enumerable.Range(0, 2000).Average(_ => CombatRewardCalculator.RollHuntReward(lvl, m.GoldBonus, m.XpBonus).Xp));

}

}

(double Turns, double Lost, double Lose) Fight(int level, int weapon, int wTier, int amulet, int aTier, Mon[] pool, double startHp = 1.0, bool ability = true, int nPer = 0)
{
    int per = nPer > 0 ? nPer : N;
    double turns = 0, lost = 0, lose = 0; int n = 0;
    foreach (string cls in Classes)
    {
        var pas = ClassPassives.For(cls);
        // La misma cuenta del juego (PlayerCombatProfileCalculator): arma con la sinergia de su clase y DESPUÉS el encantamiento.
        int wd = Enchantments.Apply(ClassWeaponSynergy.ApplyBonus(weapon, cls, Family(cls)), wTier);
        int ad = Enchantments.Apply(amulet, aTier);
        var prof = new CombatantProfile(CombatStats.TotalAttack(level, wd), CombatStats.TotalDefense(level, ad), cls, pas);
        int maxHp = (int)Math.Round((100 + 15 * (level - 1)) * pas.MaxHpMultiplier);
        int hp0 = Math.Max(1, (int)Math.Round(maxHp * startHp));
        for (int i = 0; i < per * pool.Length; i++)
        {
            var m = pool[i % pool.Length];
            int mHp = Random.Shared.Next(m.HpMin, m.HpMax + 1), mDmg = Random.Shared.Next(m.DmgMin, m.DmgMax + 1);
            int hp = hp0, t = 0; var st = default(AbilityState);
            while (true)
            {
                t++;
                bool ab = ability && CombatTurnResolver.CheckAbility(cls, st) == AbilityAvailability.Ready;
                var r = CombatTurnResolver.ResolveTurn(prof, st, ab ? PlayerAction.Ability : PlayerAction.Attack, hp, maxHp, mHp, mDmg);
                hp = r.PlayerHpAfter; mHp = r.MonsterHpAfter; st = r.StatusAfter;
                if (mHp <= 0) break;
                if (hp <= 0) { lose++; break; }
            }

            turns += t; lost += (double)(maxHp - Math.Max(0, hp)) / maxHp; n++;
        }
    }

    return (turns / n, 100.0 * lost / n, 100.0 * lose / n);
}

record Mon(string Name, int HpMin, int HpMax, int DmgMin, int DmgMax, int GoldBonus, int XpBonus);
